@ResidualSLT.integral_Ioo_split : ∀ {g : ℝ → ℝ} {a b c : ℝ},
  a ≤ b →
    b ≤ c →
      MeasureTheory.IntegrableOn g (Set.Ioo a c) MeasureTheory.volume →
        ∫ (x : ℝ) in Set.Ioo a c, g x = (∫ (x : ℝ) in Set.Ioo a b, g x) + ∫ (x : ℝ) in Set.Ioo b c, g x

ResidualSLT.integral_Ioo_inv_positive : ∀ (a b : ℝ),
  0 < a → 0 < b → a ≤ b → ∫ (u : ℝ) in Set.Ioo a b, u⁻¹ = Real.log b - Real.log a

ResidualSLT.criticalLaplaceKernel_lower : ∀ (q : ℝ),
  0 < q →
    q < 1 →
      Real.exp (-2) * q ^ 2 * -Real.log q ≤
        ∫ (z : ℝ × ℝ) in ResidualSLT.cancellationRectangle, ResidualSLT.criticalLaplaceKernel q z
ResidualSLT.criticalLaplaceKernel_upper : ∀ (q : ℝ),
  0 < q →
    q < 1 →
      ∫ (z : ℝ × ℝ) in ResidualSLT.cancellationRectangle, ResidualSLT.criticalLaplaceKernel q z ≤
        q ^ 2 * (1 - 2 * √Real.pi * Real.log q)
ResidualSLT.criticalLaplace_bounds : ∀ (N : ℝ),
  Real.exp 4 ≤ N →
    Real.exp (-2) / 4 * (N ^ (-(1 / 2)) * Real.log N) ≤ ResidualSLT.criticalLaplace N ∧
      ResidualSLT.criticalLaplace N ≤ (1 + √Real.pi) * (N ^ (-(1 / 2)) * Real.log N)
ResidualSLT.criticalLaplace_two_sided : ∃ c C N₀,
  0 < c ∧
    0 < C ∧
      1 < N₀ ∧
        ∀ (N : ℝ),
          N₀ ≤ N →
            c * (N ^ (-(1 / 2)) * Real.log N) ≤ ResidualSLT.criticalLaplace N ∧
              ResidualSLT.criticalLaplace N ≤ C * (N ^ (-(1 / 2)) * Real.log N)
