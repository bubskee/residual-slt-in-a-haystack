def ResidualSLT.HasCriticalLaplaceOrder : (ℝ → ℝ) → Prop :=
fun F => ResidualSLT.CriticalLaplaceLower F ∧ ResidualSLT.CriticalLaplaceUpper F
def ResidualSLT.CriticalLaplaceLower : (ℝ → ℝ) → Prop :=
fun F => ∃ c, 0 < c ∧ ∀ᶠ (N : ℝ) in Filter.atTop, c * ResidualSLT.criticalLaplaceRate N ≤ F N
def ResidualSLT.CriticalLaplaceUpper : (ℝ → ℝ) → Prop :=
fun F => ∃ C, 0 < C ∧ ∀ᶠ (N : ℝ) in Filter.atTop, F N ≤ C * ResidualSLT.criticalLaplaceRate N
def ResidualSLT.networkPopulationLoss : MeasureTheory.Measure ℝ → ℕ → {L : ℕ} → (Fin L → ℝ) → ℝ :=
fun μ r {L} w => ∫ (x : ℝ), ResidualSLT.networkSquaredError r w x ∂μ
def ResidualSLT.networkSquaredError : ℕ → {L : ℕ} → (Fin L → ℝ) → ℝ → ℝ :=
fun r {L} w x => (Polynomial.eval x (ResidualSLT.network r (fun i => ResidualSLT.zeroExtend w i ^ r) L) - x) ^ 2
