def ResidualSLT.LocalNetworkCertificate : MeasureTheory.Measure ℝ →
  ℕ → {L : ℕ} → (Fin L → ℝ) → (Fin L → ℝ) → ((Fin L → ℝ) → NNReal) → ℝ × ℕ → Prop :=
fun μ r {L} t x ρ p =>
  ∃ ε₀,
    0 < ε₀ ∧
      ε₀ ≤ 1 ∧
        ∀ (ε : ℝ),
          0 < ε →
            ε ≤ ε₀ →
              (∀ (s : ℝ),
                  MeasureTheory.IntegrableOn (fun w => ResidualSLT.networkTargetLoss μ r t w ^ (-s)) (Metric.ball x ε)
                      (MeasureTheory.volume.withDensity fun w => ↑(ρ w)) ↔
                    s < p.1) ∧
                ¬MeasureTheory.IntegrableOn (fun w => ResidualSLT.networkTargetLoss μ r t w ^ (-p.1)) (Metric.ball x ε)
                      (MeasureTheory.volume.withDensity fun w => ↑(ρ w)) ∧
                  ResidualSLT.HasLaplaceOrder p.1 p.2 fun N =>
                    ∫ (w : Fin L → ℝ) in Metric.ball x ε,
                      Real.exp
                        (-N * ResidualSLT.networkTargetLoss μ r t w) ∂MeasureTheory.volume.withDensity fun w => ↑(ρ w)
