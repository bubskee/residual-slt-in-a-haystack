def ResidualSLT.oneActiveAssemble : (p q : ℕ) → ℝ → OneActiveAux p q → Fin (p + 1 + q) → ℝ :=

theorem ResidualSLT.oneActive_network_laplace_order : ∀ (μ : MeasureTheory.Measure ℝ) (r p q : ℕ),
  2 ≤ r →
    FinitePositiveFeatureGram μ (sparseNetworkFeatures r (p + 1 + q)) →
      ∀ (τ : ℝ),
        τ ≠ 0 →
          ∃ ε₀,
            0 < ε₀ ∧
              ε₀ ≤ 1 ∧
                ∀ (ε : ℝ),
                  0 < ε →
                    ε ≤ ε₀ →
                      ∀ (ρ : OneActiveAux p q → NNReal),
                        BoundedPositivePriorOn ρ (Metric.ball 0 ε) MeasureTheory.volume →
                          HasLaplaceOrder (oneActiveLearningPair r p q).1 (oneActiveLearningPair r p q).2 fun N =>
                            ∫ (w : OneActiveAux p q) in Metric.ball 0 ε,
                              Real.exp
                                (-N *
                                  oneActivePopulationLoss μ r (τ ^ r)
                                    (oneActiveAssemble p q τ w)) ∂MeasureTheory.volume.withDensity fun w => ↑(ρ w) :=
fun μ r p q hr hμ τ hτ =>
  oneActive_laplace_of_local_comparison r p q hr
    (fun w => oneActivePopulationLoss μ r (τ ^ r) (oneActiveAssemble p q τ w))
    (Continuous.comp
      (oneActivePopulationLoss_continuous μ r (p + 1 + q) hr
        (Decidable.byContradiction fun a => ResidualSLT.oneActive_network_laplace_order._proof_1_1✝ r p q a) hμ (τ ^ r))
      (oneActiveAssemble_continuous p q τ))
    (oneActive_network_local_comparison μ r p q hr hμ τ hτ)
