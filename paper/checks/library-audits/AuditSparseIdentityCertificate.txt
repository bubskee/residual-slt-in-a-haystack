def ResidualSLT.sparseNetworkFeatureCount : ℕ → ℕ → ℕ :=
fun r L => (r ^ L - 1) / (r - 1)
def ResidualSLT.sparseNetworkFeatures : (r L : ℕ) → Fin (ResidualSLT.sparseNetworkFeatureCount r L) → ℝ → ℝ :=
fun r L j x => x ^ (r + ↑j * (r - 1))
def ResidualSLT.sparseNetworkCoefficients : (r : ℕ) →
  {L : ℕ} → (Fin L → ℝ) → Fin (ResidualSLT.sparseNetworkFeatureCount r L) → ℝ :=
fun r {L} w j =>
  (ResidualSLT.network r (fun i => ResidualSLT.zeroExtend w i ^ r) L - Polynomial.X).coeff (r + ↑j * (r - 1))
def ResidualSLT.identityLearningPair : ℕ → ℕ → ℝ × ℕ :=
fun r L =>
  if Even r then (↑L / (2 * ↑r), 1)
  else if L < r then (↑L / (2 * ↑r), 1) else if L = r then (1 / 2, 2) else ((↑L + ↑r) / (4 * ↑r), 1)
def ResidualSLT.HasLaplaceOrder : ℝ → ℕ → (ℝ → ℝ) → Prop :=
fun ℓ m Z =>
  0 < m ∧
    ∃ c C N₀,
      0 < c ∧
        0 < C ∧
          1 < N₀ ∧ ∀ (N : ℝ), N₀ ≤ N → c * ResidualSLT.laplaceRate ℓ m N ≤ Z N ∧ Z N ≤ C * ResidualSLT.laplaceRate ℓ m N
def ResidualSLT.laplaceRate : ℝ → ℕ → ℝ → ℝ :=
fun ℓ m N => N ^ (-ℓ) * Real.log N ^ (m - 1)
def ResidualSLT.networkPopulationLoss : MeasureTheory.Measure ℝ → ℕ → {L : ℕ} → (Fin L → ℝ) → ℝ :=
fun μ r {L} w => ∫ (x : ℝ), ResidualSLT.networkSquaredError r w x ∂μ
def ResidualSLT.networkSquaredError : ℕ → {L : ℕ} → (Fin L → ℝ) → ℝ → ℝ :=
fun r {L} w x => (Polynomial.eval x (ResidualSLT.network r (fun i => ResidualSLT.zeroExtend w i ^ r) L) - x) ^ 2
