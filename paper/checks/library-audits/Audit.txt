def ResidualSLT.CoeffMajorized : Polynomial ℝ → Polynomial ℝ → Prop :=
fun p q => ∀ (k : ℕ), |p.coeff k| ≤ q.coeff k
def ResidualSLT.networkRemainder : ℕ → ℕ → (ℕ → ℝ) → Polynomial ℝ :=
fun r n u => ResidualSLT.network r u n - Polynomial.X - Polynomial.C (∑ i ∈ Finset.range n, u i) * Polynomial.X ^ r
def ResidualSLT.signedRemainderEnvelope : ℕ → ℕ → ℝ :=
fun r x => Nat.brecOn x (ResidualSLT.signedRemainderEnvelope._f r)
def ResidualSLT.weightSquareSum : ℕ → {L : ℕ} → (Fin L → ℝ) → ℝ :=
fun r {L} w => ∑ i, (w i ^ r) ^ 2
def ResidualSLT.cancellationModel : ℕ → {L : ℕ} → (Fin L → ℝ) → ℝ :=
fun r {L} w => ResidualSLT.weightPowerSum r w ^ 2 + ResidualSLT.weightSquareSum r w ^ 2
