ResidualSLT.criticalOuterSection_eq : ∀ (s t : ℝ),

ResidualSLT.criticalSectorMellin_integrable : ∀ s < 1 / 2,
  (∀ (u : ℝ), 0 < u → MeasureTheory.IntegrableOn (fun x => (u ^ 4) ^ (-s)) (Set.Ioo 0 u) MeasureTheory.volume) ∧
    (∀ (t : ℝ),
        0 < t → MeasureTheory.IntegrableOn (fun u => (u ^ 2 * t ^ 2) ^ (-s)) (Set.Ioo 0 t) MeasureTheory.volume) ∧
      MeasureTheory.IntegrableOn (ResidualSLT.criticalCoreSection s) (Set.Ioo 0 1) MeasureTheory.volume ∧
        MeasureTheory.IntegrableOn (ResidualSLT.criticalOuterSection s) (Set.Ioo 0 1) MeasureTheory.volume
ResidualSLT.criticalSectorMellin_eq : ∀ s < 1 / 2,

  ResidualSLT.criticalSectorLoss z ≤ z.1 ^ 2 * (z.2 ^ 2 + z.1 ^ 2) ∧

ResidualSLT.criticalSectorLoss_outer : ∀ (u t : ℝ),
