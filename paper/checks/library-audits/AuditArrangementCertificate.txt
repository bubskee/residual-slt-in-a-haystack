def ResidualSLT.HasLaplaceOrder : ℝ → ℕ → (ℝ → ℝ) → Prop :=
fun ℓ m Z =>
  0 < m ∧
    ∃ c C N₀, 0 < c ∧ 0 < C ∧ 1 < N₀ ∧ ∀ (N : ℝ), N₀ ≤ N → c * laplaceRate ℓ m N ≤ Z N ∧ Z N ≤ C * laplaceRate ℓ m N
