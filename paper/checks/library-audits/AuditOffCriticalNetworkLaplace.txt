fun r {L} w x => (Polynomial.eval x (ResidualSLT.network r (fun i => ResidualSLT.zeroExtend w i ^ r) L) - x) ^ 2

def ResidualSLT.HasPowerLaplaceOrder : ℝ → (ℝ → ℝ) → Prop :=
fun ℓ F => ResidualSLT.PowerLaplaceLower ℓ F ∧ ResidualSLT.PowerLaplaceUpper ℓ F
def ResidualSLT.PowerLaplaceLower : ℝ → (ℝ → ℝ) → Prop :=
fun ℓ F => ∃ c, 0 < c ∧ ∀ᶠ (N : ℝ) in Filter.atTop, c * N ^ (-ℓ) ≤ F N
def ResidualSLT.PowerLaplaceUpper : ℝ → (ℝ → ℝ) → Prop :=
fun ℓ F => ∃ C, 0 < C ∧ ∀ᶠ (N : ℝ) in Filter.atTop, F N ≤ C * N ^ (-ℓ)
